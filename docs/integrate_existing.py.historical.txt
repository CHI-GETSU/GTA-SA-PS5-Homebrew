from pathlib import Path
root=Path(__file__).resolve().parent/'sahin/launcher'
p=root/'src/main.cpp'; s=p.read_text()
s=s.replace('#include "library/game_install.hpp"','#include "library/game_install.hpp"\n#include "library/adopt_gta.hpp"')
old='    const bool installed = kSingleGame[0] && elevated && single_game_installed(app_root, storage_root);'
assert old in s
s=s.replace(old,'''    if (kSingleGame[0] && elevated) {
        store::library::InstallPlan existing;
        std::string error;
        if (single_game_plan(app_root,storage_root,existing))
            (void)store::library::adopt_gta(existing,{storage_root,"/data/prospero-win"},error);

    }
'''+old)
